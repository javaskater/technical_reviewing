```python
>>> import torch
>>> a = torch.arange(4.)
>>> a
tensor([0., 1., 2., 3.])
>>> b = torch.reshape(a, (2,2))
>>> b
tensor([[0., 1.],
        [2., 3.]]) # first the columns and then the rows
>>> c = b.transpose(-1, -2)

tensor([[0., 2.],
```  
